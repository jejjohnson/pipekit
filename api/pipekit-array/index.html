    pipekit-array¶

    Array-API operators on top of pipekit. One operator runs on numpy, JAX, CuPy, PyTorch, or dask via array_namespace(x) dispatch — see the design doc for the full architecture, ADRs, and the v0.1 operator catalogue.

    Phase A of the v0.1 implementation has landed: namespace dispatch + reductions + stack/concat combinators. Phases B–D add geometry, QC, and the trained-model wrapper ModelOp.

    array_namespace¶

    pipekit_array._namespace ¶

    Array API namespace dispatch shim.

    The single point of backend dispatch for every pipekit-array operator. array_namespace(*xs) returns the Array API namespace (numpy, jax.numpy, torch via array-api-compat, …) shared by all input arrays.

    Tries the optional array-api-compat library first — without it, PyTorch tensors don't expose __array_namespace__ natively and torch backends won't dispatch. Falls back to native x.__array_namespace__() lookup if the compat shim isn't installed.

    See docs/design/architecture.md §4 and ADR A1 for the rationale.

    def array_namespace(*xs: Any) -> Any:
    """Return the Array API namespace shared by all input arrays.

    Args:
        *xs: One or more array-like objects. All must share the same
            Array API backend.

    Returns:
        The module-like namespace object (e.g. `numpy`, `jax.numpy`,
        `array_api_compat.torch`).

    Raises:
        TypeError: If no input implements the Array API, or if the
            inputs span more than one backend. The error message is
            consistent whether or not `array-api-compat` is installed.
    """
    compat_ns = _compat_namespace()

    if compat_ns is not None:
        try:
            return compat_ns(*xs)
        except TypeError as exc:
            # array-api-compat's error messages differ from our
            # contract; re-raise with stable wording so tests and
            # users see a consistent shape.
            msg = str(exc)
            if "Multiple namespaces" in msg:
                names = sorted({type(x).__module__.split(".")[0] for x in xs})
                raise TypeError(_multi_backend_msg(names)) from exc
            raise TypeError(_NO_BACKEND_MSG) from exc

    namespaces: set[Any] = set()
    for x in xs:
        ns = getattr(x, "__array_namespace__", None)
        if ns is None:
            raise TypeError(
                f"Input of type {type(x).__name__!r} does not implement "
                f"the Array API (`__array_namespace__` missing). " + _NO_BACKEND_MSG
            )
        namespaces.add(ns())
    if len(namespaces) == 0:
        raise TypeError(_NO_BACKEND_MSG)
    if len(namespaces) > 1:
        names = sorted({ns.__name__ for ns in namespaces})
        raise TypeError(_multi_backend_msg(names))
    return next(iter(namespaces))

    class MeanScalar(Operator):
    """Reduce an array to a scalar (or per-axis means) via `xp.mean`.

    Defaults to a full reduction (`axis=None`) yielding a scalar of
    rank 0. Output dtype matches the input's floating dtype per the
    Array API `mean` contract.

    Args:
        axis: Axis or tuple of axes to reduce over. `None` reduces
            all axes.
        keepdims: If True, retains the reduced axes with size 1.

    Raises:
        ValueError: From `compute_output_signature` when ``axis`` is out
            of range for the input signature's rank (``xp.mean`` would
            raise the backend-native equivalent at apply time).

    Example::

        out = MeanScalar()(image)               # scalar
        out = MeanScalar(axis=(-2, -1))(image)  # per-band mean
        out = MeanScalar(axis=0, keepdims=True)(image)  # shape (1, H, W)
    """

    #: saved states name the module it had before the rename (pipekit-array 0.0.3)
    __former_modules__: ClassVar[tuple[str, ...]] = ("pipekit_array.reduce",)

    def __init__(
        self,
        axis: int | tuple[int, ...] | None = None,
        keepdims: bool = False,
    ) -> None:
        # a tuple of axes reloads from JSON as a list, which `xp.mean` rejects
        self.axis = tuple(axis) if isinstance(axis, list) else axis
        self.keepdims = keepdims

    def _apply(self, x: Any) -> Any:
        xp = array_namespace(x)
        return xp.mean(x, axis=self.axis, keepdims=self.keepdims)

    def get_config(self) -> dict[str, Any]:
        return {"axis": self.axis, "keepdims": self.keepdims}

    def compute_output_signature(self, input_signature: Any) -> Any:
        if input_signature is None:
            return None

        input_sig: Signature = input_signature
        n_dims = len(input_sig.dims)

        def _validate(axis: int) -> int:
            """Return positive axis index; raise on out-of-range."""
            if n_dims == 0 or axis < -n_dims or axis >= n_dims:
                raise ValueError(
                    f"MeanScalar.axis={axis!r} is out of range for input "
                    f"signature with {n_dims} dim(s); xp.mean would raise."
                )
            return axis % n_dims

        if self.axis is None:
            axes_to_reduce: tuple[int, ...] = tuple(range(n_dims))
        elif isinstance(self.axis, int):
            axes_to_reduce = (_validate(self.axis),)
        else:
            axes_to_reduce = tuple(_validate(a) for a in self.axis)

        if self.keepdims:
            new_dims = tuple(
                (name, 1) if i in axes_to_reduce else (name, size)
                for i, (name, size) in enumerate(input_sig.dims)
            )
        else:
            new_dims = tuple(
                (name, size)
                for i, (name, size) in enumerate(input_sig.dims)
                if i not in axes_to_reduce
            )
        return Signature(dims=new_dims, dtype=input_sig.dtype)

    class ConcatenateAlong(Operator):
    """Concatenate a list-of-arrays along an existing axis via `xp.concat`.

    `xp.concat` is the Array API standardised name; numpy 2.0+
    exposes it as `concat` alongside the legacy `concatenate` alias.
    All input arrays must share shape (except along `axis`), dtype,
    and backend.

    Args:
        axis: Existing axis to concatenate along, for raw arrays. Default 0.
        dim: For named carriers instead: the dim to join along
            (`pipekit_array.concat`). ``axis`` and ``dim`` are mutually
            exclusive.

    Raises:
        ValueError: From ``_apply`` when the input sequence is empty
            (also via `array_namespace` when the inputs mix backends).

    Example::

        out = ConcatenateAlong(axis=0)([arr_a, arr_b])  # shape (2*N, H, W)
    """

    def __init__(self, axis: int | None = None, dim: str | None = None) -> None:
        if axis is not None and dim is not None:
            raise ValueError(
                "ConcatenateAlong takes axis= (raw arrays) or dim=, not both."
            )
        self.axis = 0 if axis is None and dim is None else axis
        self.dim = dim

    def _apply(self, xs: Sequence[Any]) -> Any:
        if len(xs) == 0:
            raise ValueError("ConcatenateAlong requires at least one input array.")
        if self.dim is not None:
            return dims.concat(list(xs), self.dim)
        xp = array_namespace(*xs)
        return xp.concat(list(xs), axis=self.axis)

    def get_config(self) -> dict[str, Any]:
        if self.dim is not None:
            return {"dim": self.dim}
        return {"axis": self.axis}

    Named arrays¶

    pipekit_array.dimarray ¶

    DimArray — a raw array with named dims, and its reference adapter.

    What a geo carrier becomes when a kernel consumes one of its spatial dims (per-column statistics consume row and keep col), and the plain named carrier for tests and for libraries with none of their own (PK-P7, decided 23 Oct 2026).

    • It holds a raw array (NumPy, JAX or dask), one name per axis, and the static labels of any dim (band names, level values).
    • It is a JAX pytree whose only leaf is the data: dims and labels are static, so a new set of values never recompiles. Under a bare jax.vmap (in_axes=0) the mapped dim is the first one: it is dropped inside and comes back as "batch" (vmapped_dims); to map a named dim, use pipekit_array.map_over.
    • It does not hold a georeference, coordinates beyond labels, or attrs, and does not do alignment, arithmetic or selection by value; DimArray.to_xarray crosses to xarray when an analysis needs those.

    DimArrayAdapter is the reference pipekit.NamedArrayAdapter: the test kit (pipekit_array.testing) compares every other adapter against it.

    @dataclasses.dataclass(frozen=True, eq=False)
class DimArray:
    """A raw array with one name per axis and static labels on some dims.

    Args:
        data: A NumPy, JAX or dask array.
        dims: One unique name per axis of ``data``.
        labels: Static labels per dim, e.g. ``{"band": ("B11", "B12")}``;
            each has one entry per index of its dim. Dims left out have
            none. Values are stored as tuples.

    Raises:
        ValueError: The dims do not match the data's rank, repeat a name,
            or a label tuple's length differs from its dim's size.

    Example:
        >>> import numpy as np
        >>> mu = DimArray(np.zeros((3, 2)), ("col", "band"), {"band": ["B11", "B12"]})
        >>> mu.dims, mu.shape, mu.labels["band"]
        (('col', 'band'), (3, 2), ('B11', 'B12'))
    """

    data: Any
    dims: tuple[str, ...]
    labels: Mapping[str, tuple[Any, ...]] = dataclasses.field(default_factory=dict)

    def __post_init__(self) -> None:
        dims = tuple(self.dims)
        shape = tuple(self.data.shape)
        if len(dims) != len(shape):
            raise ValueError(f"DimArray dims {dims} for data of shape {shape}.")
        if len(set(dims)) != len(dims):
            raise ValueError(f"DimArray dims must be unique, got {dims}.")
        labels: dict[str, tuple[Any, ...]] = {}
        for dim, values in dict(self.labels).items():
            if dim not in dims:
                raise ValueError(f"labels for {dim!r}, which is not in dims {dims}.")
            entries = tuple(values)
            size = shape[dims.index(dim)]
            if len(entries) != size:
                raise ValueError(
                    f"{len(entries)} labels for dim {dim!r} of size {size}."
                )
            labels[dim] = entries
        object.__setattr__(self, "dims", dims)
        object.__setattr__(self, "labels", labels)

    @property
    def shape(self) -> tuple[int, ...]:
        """The data's shape."""
        return tuple(self.data.shape)

    @property
    def dtype(self) -> Any:
        """The data's dtype."""
        return self.data.dtype

    def __array__(self, dtype: Any = None, copy: bool | None = None) -> Any:
        """The data as a NumPy array, so ``np.asarray(d)`` works (dims are dropped)."""
        import numpy as np

        if copy is None:
            return np.asarray(self.data, dtype=dtype)
        # copy=True always copies; copy=False raises if a copy is needed
        return np.array(self.data, dtype=dtype, copy=copy)

    def to_xarray(self) -> Any:
        """An ``xr.DataArray`` with the same data and dims; labels become coordinates.

        Raises:
            ImportError: xarray is not installed (``pipekit-array[xarray]``).
        """
        try:
            import xarray as xr
        except ImportError as exc:
            raise ImportError(_XARRAY_HINT) from exc
        coords = {dim: (dim, list(values)) for dim, values in self.labels.items()}
        return xr.DataArray(self.data, dims=self.dims, coords=coords)

    def vmapped_dims(dims: tuple[str, ...], shape: Any) -> tuple[str, ...]:
    """Static dims matched to leaves ``jax.vmap`` added or removed an axis from.

    pipekit_array.dims ¶

    Dim-level functions: selection, reduction, stacking and mapping by dim name.

    Each is written once, over the carrier's registered pipekit.NamedArrayAdapter, and behaves the same on every registered carrier: a DimArray, an xr.DataArray ([xarray]), a geoarray carrier. The rule for joining this list (GA-Q3): a function never reads coordinate values or geometry. resample, groupby and rolling stay xarray's; reprojection and sampling at points stay in geoarray.

    • isel, sel — by position, by exact label; an integer or a single label drops its dim.
    • transpose — reorder dims by name.
    • reduce, argreduce — over named dims; reduce(..., where=valid) leaves masked entries out (a padded batch, nodata).
    • take_along — gather along a dim with an index carrier (argreduce then take_along is a max-NDVI composite).
    • map_over — a function on each item along a dim.
    • stack, concat — a new dim, or along an existing one.
    • stack_items — items that may each carry their own geometry (patches, per-frame grids) along a new leading dim; stack where that is the same thing.
    • combine — an element-wise function of carriers with the same dims.

    Fast paths. An adapter may provide a method of the same name (the xarray adapter's reduce stays lazy on dask; geoarray's map_over is one jax.vmap over data and transform rows). It is called with the same arguments instead of the generic path; the test kit (pipekit_array.testing) proves the two agree.

    The Operator forms live in pipekit_array.dim_operators.

    def from_rows(rows: Any, layout: RowLayout, *, feature_dim: str | None = None) -> Any:
    """A carrier like ``layout.template`` from a table of rows.

    Args:
        rows: ``(n_samples, n_features)`` gives back the original feature
            dims; ``(n_samples, k)`` with ``feature_dim`` gives one new dim
            of size ``k`` in their place (PCA components, class
            probabilities); ``(n_samples,)`` gives the sample dims only
            (a label or a score per sample).
        layout: From `to_rows`.
        feature_dim: The name of the new feature dim, for a table whose
            columns are not the original features.

    Returns:
        A carrier with ``layout``'s dims in their original order, the new
        feature dim (if any) where the first feature dim was, rewrapped by
        the template's adapter.

    Raises:
        ValueError: The rows do not match the layout, or a new feature dim
            is needed and ``feature_dim`` is missing or already a dim.

    Example:
        >>> import numpy as np
        >>> from pipekit_array import DimArray
        >>> scene = DimArray(np.arange(24.0).reshape(2, 3, 4), ("band", "y", "x"))
        >>> rows, layout = to_rows(scene, sample_dims=("y", "x"))
        >>> from_rows(rows[:, :1] * 0, layout, feature_dim="score").dims
        ('score', 'y', 'x')
    """
    template = layout.template
    if template is None:
        raise ValueError("from_rows needs a layout from to_rows (it has no template).")
    adapter = adapter_of(template, "from_rows")
    shape = tuple(int(n) for n in rows.shape)
    if len(shape) not in (1, 2) or shape[0] != layout.n_samples:
        raise ValueError(
            f"from_rows: rows of shape {shape} do not fit {layout.n_samples} samples "
            f"over {layout.sample_dims}."
        )
    sample_sizes = tuple(layout._size(d) for d in layout.sample_dims)
    if len(shape) == 1:
        block_dims = layout.sample_dims
        block_shape = sample_sizes
        order = layout.sample_dims
    elif feature_dim is None:
        if shape[1] != layout.n_features:
            raise ValueError(
                f"from_rows: {shape[1]} columns are not the {layout.n_features} "
                f"features over {layout.feature_dims}; pass feature_dim= to name "
                "the new columns."
            )
        block_dims = (*layout.sample_dims, *layout.feature_dims)
        block_shape = (*sample_sizes, *(layout._size(d) for d in layout.feature_dims))
        order = layout.dims
    else:
        if feature_dim in layout.dims:
            raise ValueError(
                f"from_rows: feature_dim {feature_dim!r} is already a dim of "
                f"{layout.dims}."

        block_dims = (*layout.sample_dims, feature_dim)
        block_shape = (*sample_sizes, shape[1])
        order = _with_new_feature_dim(layout, feature_dim)

    block = xp.reshape(rows, block_shape)
    axes = tuple(block_dims.index(d) for d in order)
    if axes != tuple(range(len(axes))):
        block = xp.permute_dims(block, axes)
    return adapter.rewrap(template, block, tuple(order))

    pipekit_array.einx ¶

    A named-dims einx bridge on any carrier (geostack OP-O2).

    einx <https://github.com/fferflo/einx>_ writes tensor algebra as patterns ("band y x -> y x band"). Here the pattern's axis names are the carrier's dim names: each input is matched to its slot by name, einx runs on the data's own backend (NumPy, JAX under jit), and the result is rewrapped through the first input's adapter with the output slot's dims, so it keeps the labels, coordinates and (for geoarray) the grid of the dims that survive:

    rearrange("band y x -> y x band", scene)            # transpose by name
reduce("time band y x -> band y x", cube, op="mean") # over time
einsum("time band, band k -> time k", cube, w)       # contract band
repeat("y x -> member y x", field, member=8)         # broadcast

    einsum, reduce and repeat match input slots to dims by name, in any order. rearrange also splits and merges dims ("(a b)"), so its input slot is read positionally: it lists the input's dims in order, a group standing for one dim that is split. A merged output group (a b) becomes one dim named a_b.

    dask is not an einx backend: a dask-backed input raises pipekit.BackendError naming .compute() rather than computing it silently. einx is imported lazily: install the [einx] extra.

    pipekit_array.batching ¶

    Fixed-size batches of named carriers, with a mask of the real entries.

    batch groups a stream of carriers into Padded batches of one size, so a jitted step compiles once for every batch, the short last one included:

    items ──batch(size=4, dim="sample", pad="last")──▶ Padded
  i₀ i₁ i₂ i₃ i₄ i₅                                 value  (sample, *item_dims)
                                                    valid  (sample,) bool
  batch 1: i₀ i₁ i₂ i₃    valid = T T T T
  batch 2: i₄ i₅ i₅ i₅    valid = T T F F   ← copies of the last real item

    Padding copies the last real item: its values are finite and in distribution, so a masked loss and its gradient stay finite and a normalisation over the batch sees real statistics. Reduce with reduce(out, "mean", "sample", where=padded.valid) to leave the padding out.

    def batch(
    items: Iterable[Any], *, size: int, dim: str, pad: PadPolicy = "last"
) -> Iterator[Padded]:
    """Group named carriers into `Padded` batches of ``size`` along ``dim``.

    Args:
        items: Named carriers with the same dims (each registered with
            pipekit's adapter registry), consumed lazily.
        size: Items per batch; must be positive.
        dim: The new leading batch dim (``"sample"``, ``"patch"``,
            ``"member"``); must not be a dim of the items.
        pad: What to do with a short last batch:

            - ``"last"`` (default): pad it to ``size`` with copies of its
              last real item, ``valid`` False for the copies, so every
              batch has one shape;
            - ``"none"``: yield it short, all valid;
            - ``"drop"``: drop it (``drop_remainder`` in tf.data / Grain).

    Yields:
        One `Padded` per ``size`` items, plus the last short batch per
        ``pad``. Each ``value`` is `stack_items` of its items.

    Raises:
        ValueError: ``size`` is not positive, ``pad`` is not a policy, or
            the items' dims differ (from `stack_items`).

    Example:
        >>> import numpy as np
        >>> from pipekit_array import DimArray
        >>> items = [DimArray(np.ones((2, 2)), ("y", "x")) for _ in range(5)]
        >>> def sizes(pad):
        ...     return [len(b.valid) for b in batch(items, size=2, dim="s", pad=pad)]
        >>> sizes("last"), sizes("none"), sizes("drop")
        ([2, 2, 2], [2, 2, 1], [2, 2])
    """
    if size <= 0:
        raise ValueError(f"batch: size must be positive, got {size}.")
    if pad not in PAD_POLICIES:
        raise ValueError(f"batch: pad must be one of {PAD_POLICIES}, got {pad!r}.")
    group: list[Any] = []
    for item in items:
        group.append(item)
        if len(group) == size:
            yield _padded(group, dim, 0)
            group = []
    if group and pad != "drop":
        yield _padded(group, dim, size - len(group) if pad == "last" else 0)

    Kernels¶

    pipekit_array.kernel ¶

    KernelSpec and KernelOp — one L1 kernel, written once, on every carrier.

    A shapes-only kernel declares the axes it consumes and produces with a generalized-ufunc signature ("(c)->()", "(r,c),(c),(c,d)->(r)"), the convention NumPy (np.vectorize), JAX (jnp.vectorize) and xarray (apply_ufunc core dims) already share. KernelOp wraps the kernel once and runs it:

    raw arrays        → the kernel, called as is
named carriers    → core dims moved last by name, the kernel on the
                    data, the result rewrapped by the adapter
                    (NumPy eagerly, JAX traced under jit)
an adapter with   → that adapter's own path (the xarray adapter:
``apply_kernel``    one ``apply_ufunc``, lazy on dask)

    Kernel conventions:

    1. Consumed axes last. Core dims arrive as the trailing axes.
    2. Broadcast over leading axes, or set vectorize=True and pay for np.vectorize / jnp.vectorize.
    3. Non-carrier inputs are keyword parameters (a background mean, a threshold), passed whole and never mapped. Geometry derived from the carrier (geometry=) is passed the same way; geometry that varies per pixel has the carrier's dims and is a positional input instead.
    4. Write against the array namespace (xp.linalg.solve) to run on every backend. A kernel that needs a JAX-only library (lineax) declares backends=("jax",) (spike finding F3).

    Rules from the spike (geotoolz#516):

    • Outputs keep the first input's dim order (F2): a gufunc puts core dims last, so each output is transposed back, with dims new to the output last.
    • geometry= needs one geometry per call (F4): on a carrier whose adapter reports batch_dims (per-item transforms), KernelOp raises and names map_over.
    • An output-only core dim declares its size (F5): KernelSpec.sizes maps it to an input core dim, so a lazy backend can plan the output.

    Carriers of different kinds combine by dim name and size (a geo carrier "band row col" with a DimArray "col band"); grid checks (check_compatible) run between carriers of the same kind.

        self,

        spec: KernelSpec | str,

        geometry: dict[str, Callable[[Any], Any]] | None = None,
        params: Mapping[str, Any] | None = None,
        **kwargs: Any,

        letters = set(self.spec.letters)
        if set(core) != letters:

                f"got {sorted(core)}."

                "pass a kernel parameter named 'geometry' as params={'geometry': ...}."

        if clash:
            raise ValueError(
                f"{clash} are both parameters and derived-geometry keywords; "
                "the kernel can take each name once."

                f"got {len(carriers)}."

class KernelSpec:
    """A generalized-ufunc signature over abstract core-dim letters.

    Args:
        signature: E.g. ``"(),()->()"`` (element-wise), ``"(c)->()"``
            (consumes one axis), ``"(r,c)->(c),(c,d)"`` (two outputs).
        backends: The backends the kernel is written for, from ``"numpy"``,
            ``"jax"``, ``"dask"``, ``"xarray"``.
        vectorize: The kernel handles only its core dims, and is mapped over
            the rest with ``np.vectorize`` / ``jnp.vectorize``.
        sizes: Output-only core letters and the input letter whose size each
            has, e.g. ``(("d", "c"),)``. Pairs, not a dict, so the spec stays
            hashable.
        dtypes: Each output's dtype, for output signatures and lazy
            backends: a dtype name (``"bool"``) or ``"same"`` for the first
            input's. Empty (the default) leaves it unknown in signatures;
            dask then finds it by calling the kernel once.

    Raises:
        ValueError: The signature does not parse, ``sizes`` names a letter
            that is not output-only or an input letter that does not exist,
            or ``dtypes`` is not one entry per output.

    Example:
        >>> spec = KernelSpec("(r,c)->(c),(c,d)", sizes=(("d", "c"),))
        >>> spec.inputs, spec.outputs
        ((('r', 'c'),), (('c',), ('c', 'd')))
    """

    signature: str
    backends: tuple[str, ...] = ("numpy", "jax", "dask", "xarray")
    vectorize: bool = False
    sizes: tuple[tuple[str, str], ...] = ()
    dtypes: tuple[str, ...] = ()

    def __post_init__(self) -> None:
        if not _SIGNATURE.match(self.signature):
            raise ValueError(f"Not a gufunc signature: {self.signature!r}.")
        object.__setattr__(self, "backends", tuple(self.backends))
        object.__setattr__(self, "sizes", tuple(tuple(p) for p in self.sizes))
        object.__setattr__(self, "dtypes", tuple(map(str, self.dtypes)))
        if self.dtypes and len(self.dtypes) != len(self.outputs):
            raise ValueError(
                f"dtypes: {len(self.dtypes)} entries for {len(self.outputs)} output(s)."
            )
        in_letters = {c for group in self.inputs for c in group}
        out_only = {c for group in self.outputs for c in group} - in_letters
        declared = {out for out, _ in self.sizes}
        for out, source in self.sizes:
            if out not in out_only:
                raise ValueError(f"sizes: {out!r} is not an output-only letter.")
            if source not in in_letters:
                raise ValueError(f"sizes: {source!r} is not an input letter.")
        missing = sorted(out_only - declared)
        if missing:
            raise ValueError(
                f"Output-only core letter(s) {missing} need a size, e.g. "
                f"sizes=(({missing[0]!r}, '<input letter>'),)."
            )

    @functools.cached_property
    def inputs(self) -> tuple[tuple[str, ...], ...]:
        """Core letters of each input."""
        return _groups(self.signature.split("->")[0])

    @functools.cached_property
    def outputs(self) -> tuple[tuple[str, ...], ...]:
        """Core letters of each output."""
        return _groups(self.signature.split("->")[1])

    @functools.cached_property
    def letters(self) -> tuple[str, ...]:
        """Every core letter, in order of first appearance."""
        seen: dict[str, None] = {}
        for group in (*self.inputs, *self.outputs):
            for letter in group:
                seen.setdefault(letter, None)
        return tuple(seen)

    def kernel_output_dims(
    spec: KernelSpec, core: dict[str, str], input_dims: Sequence[tuple[str, ...]]

    """The dims of each output, from the inputs' dims, without running.

    input's order, with dims new to the output last (F2).

        cores = {core[c] for c in group}
        for d in dims:

        ordered = [d for d in first if d in dims] + [d for d in dims if d not in first]

    class XarrayAdapter:
    """The `pipekit.NamedArrayAdapter` for ``xr.DataArray``, with native fast paths."""

    # -- the protocol ----------------------------------------------------

    def dims(self, x: xr.DataArray) -> tuple[str, ...]:
        """``x.dims``."""
        return tuple(str(d) for d in x.dims)

    def data(self, x: xr.DataArray) -> Any:
        """``x.data``: NumPy, dask or JAX, never computed."""
        return x.data

    def labels(self, x: xr.DataArray, dim: str) -> tuple[Any, ...] | None:
        """The values of ``dim``'s 1-D index coordinate, or None."""
        if dim in x.indexes and x.coords[dim].ndim == 1:
            values = x.coords[dim].values
            # tolist() turns datetime64[ns] into integers; keep the scalars
            if values.dtype.kind in "mM":
                return tuple(values)
            return tuple(values.tolist())
        return None

    def isel(self, x: xr.DataArray, indexers: dict[str, Any]) -> xr.DataArray:
        """``x.isel``; coordinates follow the data."""
        return x.isel(indexers)

    def rewrap(self, x: xr.DataArray, data: Any, dims: tuple[str, ...]) -> xr.DataArray:
        """A DataArray around new data, keeping coords and attrs of surviving dims.

        A coordinate is kept when all its dims survive with their sizes.
        """
        sizes = dict(zip(dims, data.shape, strict=True))
        coords = {
            name: coord
            for name, coord in x.coords.items()
            if all(sizes.get(d) == x.sizes[d] for d in coord.dims)
        }
        out = xr.DataArray(data, dims=dims, coords=coords, attrs=x.attrs, name=x.name)
        # name=None lets xarray take a dask array's graph name; keep x's instead
        return out if out.name == x.name else out.rename(x.name)

    def check_compatible(self, *xs: xr.DataArray) -> tuple[xr.DataArray, ...]:
        """The inputs back if their shared coordinates are exactly equal.

        Raises:
            ValueError: Coordinates differ; xarray would otherwise inner-join.
        """
        try:
            xr.align(*xs, join="exact", copy=False)
        except ValueError as exc:
            raise ValueError(
                f"DataArray inputs are not aligned (join='exact'): {exc}"
            ) from exc
        return xs

    def signature(self, x: xr.DataArray) -> Signature:
        """Dims with their sizes, and the dtype's name."""
        dims = tuple((str(d), int(n)) for d, n in x.sizes.items())
        return Signature(dims=dims, dtype=str(x.dtype))

    # -- dim-level fast paths --------------------------------------------

    def transpose(self, x: xr.DataArray, order: tuple[str, ...]) -> xr.DataArray:
        """``x.transpose``."""
        return x.transpose(*order)

    def reduce(
        self,
        x: xr.DataArray,
        fn: str | Callable[..., Any],
        dims: tuple[str, ...],
        *,
        where: Any | None = None,
    ) -> xr.DataArray:
        """``x.reduce`` with the array-namespace function, lazy on dask.

        A name resolves to NumPy's function (``"nanmean"`` included), so NaN
        handling matches the generic path: DataArray's own ``mean`` would
        skip NaN by default. ``where`` arrives aligned to ``x``'s full shape
        (`pipekit_array.reduce` aligns it) and is passed to the function.
        """
        func = _numpy_function(fn, "reduce")
        if where is None:
            return x.reduce(func, dim=list(dims))
        return x.reduce(func, dim=list(dims), where=where)

    def argreduce(
        self, x: xr.DataArray, fn: str | Callable[..., Any], dim: str
    ) -> xr.DataArray:
        """``x.reduce`` with NumPy's ``argmax`` / ``argmin`` (or ``fn``)."""
        return x.reduce(_numpy_function(fn, "argreduce"), dim=dim)

    def take_along(self, x: xr.DataArray, index: Any, dim: str) -> xr.DataArray:
        """One entry of ``dim`` per position, lazily on dask, in ``x``'s dim order.

        An ``apply_ufunc`` over ``take_along_axis``, not a vectorized
        ``isel``: dask cannot index with a lazy indexer. Attributes are kept,
        auxiliary coordinates on ``dim`` (``quality(time, x)``) are gathered
        with the same index, and ``dim``'s own coordinate is dropped.
        """
        rest = [str(d) for d in x.dims if d != dim]
        idx = _as_dataarray(index)  # raises for an unnamed index
        index_dims = tuple(map(str, idx.dims))
        if not set(index_dims) <= set(rest):
            # the gather would broadcast over an unrelated index dim
            raise ValueError(
                f"take_along: index dims {index_dims} are not among {tuple(rest)}."
            )
        # the labels are checked on the index itself: idx drops its coordinates
        index_adapter = get_adapter(index)
        if index_adapter is not None:
            check_index_labels(self, x, index_adapter, index, index_adapter.dims(index))
        out = _gather(x, idx, dim)
        aux = {
            str(name): _gather(xr.DataArray(c.variable), idx, dim).variable
            for name, c in x.coords.items()
            if dim in c.dims and name != dim
        }
        if aux:
            out = out.assign_coords(aux)
        return out.transpose(*rest)

    def stack(self, xs: Sequence[xr.DataArray], dim: str) -> xr.DataArray:
        """A new leading dim; coordinates must be identical (never aligned)."""
        if dim in xs[0].dims:
            raise ValueError(
                f"stack: {dim!r} is already a dim of {tuple(xs[0].dims)}; use concat."
            )
        _require_same_dims(xs, "stack")
        checked = self.check_compatible(*xs)
        return xr.concat(
            checked, dim=dim, join="exact", coords="minimal", compat="identical"
        )

    def concat(self, xs: Sequence[xr.DataArray], dim: str) -> xr.DataArray:
        """Along an existing dim, joining its coordinate; others must be identical."""
        dims = _require_same_dims(xs, "concat")
        if dim not in dims:
            raise ValueError(f"concat: {dim!r} is not a dim of {dims}; use stack.")
        return xr.concat(
            list(xs), dim=dim, join="exact", coords="minimal", compat="identical"
        )

    def combine(self, fn: Callable[..., Any], *xs: xr.DataArray) -> xr.DataArray:
        """``apply_ufunc(fn, …, join="exact")``, lazy on dask."""
        checked = self.check_compatible(*xs)
        dims = tuple(checked[0].dims)
        ordered = [x.transpose(*dims) for x in checked]
        lazy = any(_is_dask(x) for x in ordered)
        return xr.apply_ufunc(
            fn,
            *ordered,
            dask="parallelized",
            join="exact",
            # dask needs the result dtype up front: read it from a one-element call
            output_dtypes=[_probe_dtype(fn, ordered)] if lazy else None,
        )

    def apply_kernel(
        self,
        kernel: Callable[..., Any],
        spec: KernelSpec,
        core: dict[str, str],
        kwargs: dict[str, Any],
        *xs: xr.DataArray,
    ) -> Any:
        """A `KernelOp` call as one ``xr.apply_ufunc``: lazy, exact, in input order.

        Core dims move last; ``kwargs`` (params and derived geometry) are
        passed whole; output-only core dims take their sizes from
        ``spec.sizes``; outputs come back in the first input's dim order (F2).

        Raises:
            BackendError: The spec does not list ``"xarray"``.
        """
        if "xarray" not in spec.backends:
            raise BackendError(
                f"the kernel declares backends {spec.backends}, not 'xarray'."
            )
        checked = self.check_compatible(*xs)
        in_dims = [[core[c] for c in group] for group in spec.inputs]
        out_dims = [[core[c] for c in group] for group in spec.outputs]
        letter_size: dict[str, int] = {}
        for x, group in zip(checked, spec.inputs, strict=True):
            for letter in group:
                letter_size[letter] = int(x.sizes[core[letter]])
        output_sizes = {core[out]: letter_size[source] for out, source in spec.sizes}
        result = xr.apply_ufunc(
            kernel,

) -> Any:
    """A `KernelOp` call as one ``xr.apply_ufunc``: lazy, exact, in input order.

    Core dims move last; ``kwargs`` (params and derived geometry) are
    passed whole; output-only core dims take their sizes from
    ``spec.sizes``; outputs come back in the first input's dim order (F2).

    Raises:
        BackendError: The spec does not list ``"xarray"``.
    """
    if "xarray" not in spec.backends:
        raise BackendError(
            f"the kernel declares backends {spec.backends}, not 'xarray'."
        )
    checked = self.check_compatible(*xs)
    in_dims = [[core[c] for c in group] for group in spec.inputs]
    out_dims = [[core[c] for c in group] for group in spec.outputs]
    letter_size: dict[str, int] = {}
    for x, group in zip(checked, spec.inputs, strict=True):
        for letter in group:
            letter_size[letter] = int(x.sizes[core[letter]])
    output_sizes = {core[out]: letter_size[source] for out, source in spec.sizes}
    result = xr.apply_ufunc(
        kernel,
        *checked,
        input_core_dims=in_dims,
        output_core_dims=out_dims,
        kwargs=kwargs,
        vectorize=spec.vectorize,
        dask="parallelized",
        join="exact",
        output_dtypes=_output_dtypes(kernel, spec, kwargs, checked, letter_size),
        dask_gufunc_kwargs={"allow_rechunk": True, "output_sizes": output_sizes},
    )
    targets = kernel_output_dims(
        spec, core, [tuple(map(str, x.dims)) for x in checked]
    )
    if len(out_dims) > 1:
        return tuple(r.transpose(*t) for r, t in zip(result, targets, strict=True))
    return result.transpose(*targets[0])

    def rewrap(self, x: xr.DataArray, data: Any, dims: tuple[str, ...]) -> xr.DataArray:
    """A DataArray around new data, keeping coords and attrs of surviving dims.
